cmake_minimum_required(VERSION 3.10)
project(online_remani_validation)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
find_package(catkin REQUIRED COMPONENTS roscpp roslib std_msgs sensor_msgs nav_msgs geometry_msgs control_msgs visualization_msgs quadrotor_msgs cv_bridge tf2_ros pcl_conversions mm_utils)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(OpenCV REQUIRED)
find_package(ompl REQUIRED)
set(P ${SNAPSHOT}/mm_remani_planner)
include_directories(${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIR} ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${OMPL_INCLUDE_DIRS})
foreach(part plan_manage plan_env mm_config path_searching traj_opt traj_utils)
  include_directories(${P}/${part}/include)
endforeach()
include_directories(${SNAPSHOT}/mm_controller/include ${SNAPSHOT}/mm_remani/include)
add_library(exp_env STATIC ${P}/plan_env/src/grid_map.cpp ${P}/plan_env/src/esdf_worker.cpp ${P}/plan_env/src/dynamic_point_buffer.cpp ${P}/plan_env/src/raycast.cpp)
target_link_libraries(exp_env ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${OpenCV_LIBRARIES})
set_target_properties(exp_env PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(exp_model STATIC ${P}/mm_config/src/mm_config.cpp)
target_link_libraries(exp_model exp_env)
add_library(exp_search STATIC ${P}/path_searching/src/kino_astar.cpp ${P}/path_searching/src/sample_mani_RRT.cpp ${P}/path_searching/src/rrt.cpp)
target_link_libraries(exp_search exp_model ${OMPL_LIBRARIES})
add_library(exp_opt STATIC ${P}/traj_opt/src/poly_traj_optimizer.cpp)
target_link_libraries(exp_opt exp_search)
add_executable(remani_planner_node ${P}/plan_manage/src/remani_planner_node.cpp ${P}/plan_manage/src/remani_replan_fsm.cpp ${P}/plan_manage/src/planner_manager.cpp ${P}/plan_manage/src/planning_visualization.cpp)
target_link_libraries(remani_planner_node exp_opt)
add_executable(mm_controller_node ${SNAPSHOT}/mm_controller/src/mm_controller_node.cpp ${SNAPSHOT}/mm_controller/src/mm_controller_fsm.cpp ${SNAPSHOT}/mm_controller/src/controller_utils.cpp)
target_link_libraries(mm_controller_node ${catkin_LIBRARIES})
add_executable(dynamic_obstacle_tracker_node ${SNAPSHOT}/mm_remani/src/dynamic_obstacle_tracker_node.cpp ${SNAPSHOT}/mm_remani/src/dynamic_obstacle_tracker_core.cpp)
target_link_libraries(dynamic_obstacle_tracker_node ${catkin_LIBRARIES} ${PCL_LIBRARIES})
include_directories(${SNAPSHOT})
add_library(online_mppi SHARED online_api.cpp)
target_link_libraries(online_mppi exp_env)
add_executable(remani_planner_node_no_wait ${P}/plan_manage/src/remani_planner_node.cpp ${P}/plan_manage/src/remani_replan_fsm.cpp ${SNAPSHOT}/planner_manager_no_wait.cpp ${P}/plan_manage/src/planning_visualization.cpp)
target_link_libraries(remani_planner_node_no_wait exp_opt)
